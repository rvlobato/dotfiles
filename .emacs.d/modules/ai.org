* gptel

#+begin_src emacs-lisp
(defun my/gptel-codex-models ()
  "Return the models the Codex CLI lists for this account, or nil.
Read `~/.codex/models_cache.json' and keep models with visibility
\"list\", sorted by the CLI's priority."
  (let ((file (expand-file-name "~/.codex/models_cache.json")))
    (when (file-readable-p file)
      (condition-case err
          (let* ((data (with-temp-buffer
                         (insert-file-contents file)
                         (json-parse-buffer :object-type 'plist
                                            :array-type 'list)))
                 (models (seq-filter
                          (lambda (m) (equal (plist-get m :visibility) "list"))
                          (plist-get data :models))))
            (mapcar (lambda (m)
                      (list (intern (plist-get m :slug))
                            :description (plist-get m :display_name)))
                    (seq-sort-by (lambda (m) (or (plist-get m :priority) 1000))
                                 #'< models)))
        (error (message "gptel: cannot read %s: %s" file
                        (error-message-string err))
               nil)))))

(my/use-package gptel
  :commands (gptel gptel-send gptel-menu gptel-rewrite gptel-add)
  :bind (("C-c RET" . gptel-send)
         ("C-c i"   . gptel)
         ("C-c I"   . gptel-menu))
  :custom
  (gptel-default-mode 'org-mode)
  (gptel-use-curl t)
  :config
  (setq gptel-backend
        (if-let* ((models (my/gptel-codex-models)))
            (gptel-make-openai-oauth "ChatGPT Plus" :models models)
          (gptel-make-openai-oauth "ChatGPT Plus")))
  (let ((models (gptel-backend-models gptel-backend)))
    (unless (memq gptel-model models)
      (setq gptel-model (car models)))))

* agent-shell

=agent-shell= talks to coding agents over ACP (Agent Client Protocol).
Gentoo installs the package and its dependencies =acp= and =shell-maker=
in =site-lisp=, so =my/use-package= does not fetch them from ELPA.

Each agent needs its ACP adapter on =PATH=, next to its CLI:
- Claude Code: =claude-agent-acp= (=@agentclientprotocol/claude-agent-acp=).
- Codex: =codex-acp= (=@agentclientprotocol/codex-acp=).
- Antigravity: =agy acp=, built into the Antigravity CLI since 1.5.
  The Gentoo package (0.88.3) still expects the old standalone
  =agy_acp_server.par=; upstream agent-shell now runs =agy acp=, so the
  command is set here.  Sign in once with =agy acp --login=.

Both use the login from the CLI (=claude=, =codex=), so no API key is set
here.  The agent process inherits the Emacs environment so that it finds
=~/.local/bin=, =pixi=, =julia= and the other tools on =PATH=.

(my/use-package agent-shell
  :commands (agent-shell agent-shell-toggle agent-shell-new-shell
             agent-shell-anthropic-start-claude-code
             agent-shell-openai-start-codex
             agent-shell-antigravity-start-agent)
  :bind (("C-c A a" . agent-shell)
         ("C-c A t" . agent-shell-toggle)
         ("C-c A n" . agent-shell-new-shell)
         ("C-c A c" . agent-shell-anthropic-start-claude-code)
         ("C-c A x" . agent-shell-openai-start-codex)
         ("C-c A g" . agent-shell-antigravity-start-agent))
  :custom
  ;; Offer the agent picker, with Claude Code preselected.
  (agent-shell-preferred-agent-config '(preselect . claude-code))
  ;; Ask whether to resume a previous session or start a new one.
  (agent-shell-session-strategy 'prompt)
  :config
  (setq agent-shell-anthropic-authentication
        (agent-shell-anthropic-make-authentication :login t))
  (setq agent-shell-openai-authentication
        (agent-shell-openai-make-authentication :login t))
  (setq agent-shell-anthropic-claude-environment
        (agent-shell-make-environment-variables :inherit-env t))
  (setq agent-shell-openai-codex-environment
        (agent-shell-make-environment-variables :inherit-env t))
  ;; TODO Remove when Gentoo agent-shell > 0.88.3 defaults to agy acp (see WORK.org).
  (setq agent-shell-antigravity-acp-command '("agy" "acp"))
  (setq agent-shell-antigravity-authentication
        (agent-shell-antigravity-make-authentication :login t))
  (setq agent-shell-antigravity-environment
        (agent-shell-make-environment-variables :inherit-env t)))
#+end_src
