#+begin_src emacs-lisp
(defvar my/notmuch-spam-folders
  '(("cbpf/INBOX"   . "~/.mail/cbpf/Junk")
    ("gmail/INBOX"  . "~/.mail/gmail/[Gmail]/Spam")
    ("proton/INBOX" . "~/.mail/proton/Spam"))
  "Alist of (SOURCE-QUERY . DEST-MAILDIR) for spam relocation.")

(defun my/notmuch--root ()
  "Root of the notmuch mail store."
  (car (process-lines "notmuch" "config" "get" "database.path")))

(defun my/notmuch--maildir-name (file flags)
  "Basename of FILE without the mbsync UID, with maildir FLAGS."
  (let ((base (replace-regexp-in-string
               ",U=[0-9]+" "" (file-name-nondirectory file))))
    (concat (replace-regexp-in-string ":2,.*\\'" "" base) ":2," flags)))

(defun my/notmuch--files-under (query dir)
  "Existing files of messages matching QUERY that lie inside DIR.
`notmuch search --output=files' lists every copy of a message, also
copies in other folders, so filter by path."
  (let ((dir (file-name-as-directory (expand-file-name dir))))
    (seq-filter (lambda (f) (and (string-prefix-p dir f) (file-exists-p f)))
                (process-lines "notmuch" "search" "--exclude=false" "--output=files" query))))

(defun my/notmuch--tagged-p (id tag)
  "Non-nil if message ID carries TAG."
  (not (string= "0" (car (process-lines
                          "notmuch" "count" "--exclude=false" (format "%s and tag:%s" id tag))))))

(defun my/notmuch-move-spam ()
  "Move messages tagged spam out of each INBOX into the account Spam folder.
Each message is trained once as spam in bogofilter and tagged bogo-spam,
so `my/notmuch-mark-ham' can undo the training."
  (interactive)
  (let ((root (my/notmuch--root)))
    (dolist (pair my/notmuch-spam-folders)
      (let ((src (expand-file-name (car pair) root))
            (dest (expand-file-name (cdr pair))))
        (dolist (id (process-lines "notmuch" "search" "--exclude=false" "--output=messages"
                                   (format "tag:spam and folder:\"%s\"" (car pair))))
          (let ((files (my/notmuch--files-under id src)))
            (when files
              (unless (my/notmuch--tagged-p id "bogo-spam")
                (call-process "bogofilter" (car files) nil nil "-s")
                (call-process "notmuch" nil nil nil "tag" "+bogo-spam" "--" id))
              (make-directory (expand-file-name "cur" dest) t)
              (dolist (f files)
                (rename-file f (expand-file-name
                                (concat "cur/" (my/notmuch--maildir-name f "S"))
                                dest)
                             t)))))))))

(defun my/notmuch--refresh-buffers ()
  "Revert all live Notmuch buffers."
  (dolist (buffer (buffer-list))
    (when (buffer-live-p buffer)
      (with-current-buffer buffer
        (when (derived-mode-p 'notmuch-search-mode
                              'notmuch-show-mode
                              'notmuch-tree-mode
                              'notmuch-hello-mode)
          (revert-buffer t t))))))

(defun my/notmuch-sync ()
  "Pull mail, push spam moves to the server, and refresh Notmuch buffers."
  (interactive)
  (unless (get-process "mail-sync")
    (let ((buf (get-buffer-create "*mail-sync*"))
          (script (executable-find "mail-sync")))
      (if script
          ;; Machine with the systemd mail-sync service: reuse its script,
          ;; which takes a lock so F6 and the hourly timer never overlap.
          (make-process
           :name "mail-sync"
           :buffer buf
           :command (list script)
           :sentinel
           (lambda (_proc event)
             (when (string-match-p "finished\\|exited" event)
               (my/notmuch--refresh-buffers)
               (message "Mail synchronized%s."
                        (if (string-match-p "finished" event)
                            "" " (with errors, see *mail-sync*)")))))
      (make-process
       :name "mail-sync"
       :buffer buf
       :command '("sh" "-c" "mbsync -a && notmuch new")
       :sentinel
       (lambda (_proc event)
         (when (string-match-p "finished" event)
           (my/notmuch-move-spam)
           (make-process
            :name "mail-sync"
            :buffer buf
            :command '("sh" "-c" "mbsync -a && notmuch new")
            :sentinel
            (lambda (_proc event)
              (when (string-match-p "finished" event)
                (my/notmuch--refresh-buffers)
                (message "Mail synchronized.")))))))))))

;; 2. BOGOFILTER: Function to train false positives (Ham)
(defun my/notmuch-mark-ham ()
  "Correct a false positive.
For each spam-tagged message in the current message (show and tree
mode) or thread (search mode): train bogofilter as ham (undoing the spam training
when the message carries bogo-spam), restore the inbox and unread tags,
and move it from the Spam folder back to the account INBOX.
Then offer to add each sender that is not yet whitelisted to the
post-new sender whitelist."
  (interactive)
  (let* ((query (cond ((derived-mode-p 'notmuch-show-mode)
                        (notmuch-show-get-message-id))
                       ((derived-mode-p 'notmuch-tree-mode)
                        (notmuch-tree-get-message-id))
                       (t (notmuch-search-find-thread-id))))
         (root (my/notmuch--root))
         (ids (process-lines "notmuch" "search" "--exclude=false" "--output=messages"
                             (format "(%s) and tag:spam" query)))
         (moved nil))
    (if (null ids)
        (message "No spam-tagged message here.")
      (dolist (id ids)
        (let ((file (seq-find #'file-exists-p
                              (process-lines "notmuch" "search" "--exclude=false" "--output=files" id))))
          (when file
            (call-process "bogofilter" file nil nil
                          (if (my/notmuch--tagged-p id "bogo-spam") "-Sn" "-n"))))
        ;; Tag first: with synchronize_flags notmuch renames the files
        ;; (drops the S flag), so query the paths afterwards.
        (call-process "notmuch" nil nil nil "tag"
                      "-spam" "-bogo-spam" "+inbox" "+unread" "--" id)
        (dolist (pair my/notmuch-spam-folders)
          (let ((inbox (expand-file-name (car pair) root)))
            (dolist (f (my/notmuch--files-under id (cdr pair)))
              (make-directory (expand-file-name "cur" inbox) t)
              (rename-file f (expand-file-name
                              (concat "cur/" (my/notmuch--maildir-name f ""))
                              inbox)
                           t)
              (setq moved t)))))
      (when moved
        (call-process "notmuch" nil nil nil "new" "--no-hooks"))
      (my/notmuch--refresh-buffers)
      (let ((added (my/notmuch--offer-whitelist ids)))
        (message "Trained %d message(s) as ham%s%s."
                 (length ids) (if moved " and moved them to INBOX" "")
                 (if added
                     (format "; whitelisted %s" (string-join added ", "))
                   ""))))))

;; Sender whitelist read by the post-new hook. The source of truth is
;; the whitelist block in tools.org; H adds to both that block and the
;; tangled file, so the next tangle keeps the entry.
(defvar my/notmuch-whitelist-org
  (expand-file-name "modules/tools.org" user-emacs-directory)
  "Org file that holds the tangled sender whitelist block.")

(defun my/notmuch--whitelist-file ()
  "Sender whitelist file in the notmuch hook directory."
  (expand-file-name "whitelist"
                    (car (process-lines "notmuch" "config" "get" "database.hook_dir"))))

(defun my/notmuch--whitelist-entries ()
  "Entries of the whitelist file, lower case, without comments."
  (let ((file (my/notmuch--whitelist-file)))
    (when (file-exists-p file)
      (with-temp-buffer
        (insert-file-contents file)
        (delete ""
                (mapcar (lambda (line)
                          (downcase (replace-regexp-in-string
                                     "#.*\\|[[:space:]\"]" "" line)))
                        (split-string (buffer-string) "\n")))))))

(defun my/notmuch--whitelisted-p (addr entries)
  "Non-nil if ADDR equals an entry of ENTRIES or lies in its domain."
  (seq-some (lambda (e)
              (or (string= e addr)
                  (string-suffix-p (concat "@" e) addr)
                  (string-suffix-p (concat "." e) addr)))
            entries))

(defun my/notmuch--sender (id)
  "Bare sender address of message ID, lower case."
  (let ((addr (car (process-lines "notmuch" "address" "--exclude=false"
                                  "--output=address" "--" id))))
    (and addr (downcase addr))))

(defun my/notmuch--add-to-whitelist (addr)
  "Append ADDR to the whitelist block in tools.org and to the hook file."
  (let ((buf (find-file-noselect my/notmuch-whitelist-org)))
    (with-current-buffer buf
      (when (buffer-modified-p)
        (user-error "Save %s first, then whitelist again"
                    (file-name-nondirectory my/notmuch-whitelist-org)))
      (save-excursion
        (goto-char (point-min))
        (unless (re-search-forward
                 "^#\\+begin_src .*:tangle .*hooks/whitelist" nil t)
          (user-error "No whitelist block in %s" my/notmuch-whitelist-org))
        (re-search-forward "^#\\+end_src")
        (beginning-of-line)
        (insert addr "\n"))
      (save-buffer)))
  (let ((file (my/notmuch--whitelist-file)))
    (with-temp-buffer
      (when (file-exists-p file)
        (insert-file-contents file))
      (goto-char (point-max))
      (unless (or (bobp) (bolp))
        (insert "\n"))
      (insert addr "\n")
      (write-region nil nil file nil 'silent))))

(defun my/notmuch--offer-whitelist (ids)
  "Offer to whitelist the senders of IDS. Return the added addresses."
  (let ((entries (my/notmuch--whitelist-entries))
        added)
    (dolist (addr (delete-dups (delq nil (mapcar #'my/notmuch--sender ids))))
      (unless (my/notmuch--whitelisted-p addr entries)
        (when (y-or-n-p (format "Whitelist sender %s? " addr))
          (my/notmuch--add-to-whitelist addr)
          (push addr added))))
    (nreverse added)))

;; Bind Ham training to 'H'
(with-eval-after-load 'notmuch
  (define-key notmuch-search-mode-map (kbd "H") #'my/notmuch-mark-ham)
  (define-key notmuch-show-mode-map (kbd "H") #'my/notmuch-mark-ham)
  (define-key notmuch-tree-mode-map (kbd "H") #'my/notmuch-mark-ham))

;; `k k s' (reverse "Mark as spam" in the tag-jump menu) only flips tags.
;; The file stays in the Spam folder, and step 4 of post-new tags it spam
;; again on the next sync.  Route that key to `my/notmuch-mark-ham', so
;; `k k s' and `H' do the same thing.
(defvar my/notmuch--tag-jump-reverse nil
  "Non-nil while `notmuch-tag-jump' builds its reverse menu.")

(defun my/notmuch--tag-jump-note-reverse (orig reverse)
  "Call ORIG with REVERSE, recording REVERSE for `notmuch-jump'."
  (let ((my/notmuch--tag-jump-reverse reverse))
    (funcall orig reverse)))

(defun my/notmuch--spam-tag-keys ()
  "Keys in `notmuch-tagging-keys' whose operations add the spam tag."
  (delq nil
        (mapcar (lambda (entry)
                  (let ((ops (nth 1 entry)))
                    (when (member "+spam" (if (symbolp ops) (symbol-value ops) ops))
                      (car entry))))
                notmuch-tagging-keys)))

(defun my/notmuch--reverse-spam-to-ham (args)
  "In the reverse tag-jump menu, make the spam keys run `my/notmuch-mark-ham'."
  (if (not my/notmuch--tag-jump-reverse)
      args
    (let ((keys (my/notmuch--spam-tag-keys)))
      (cons (mapcar (lambda (item)
                      (if (member (car item) keys)
                          (list (car item) "Not spam (H)" #'my/notmuch-mark-ham)
                        item))
                    (car args))
            (cdr args)))))

(with-eval-after-load 'notmuch-tag
  (advice-add 'notmuch-tag-jump :around #'my/notmuch--tag-jump-note-reverse)
  (advice-add 'notmuch-jump :filter-args #'my/notmuch--reverse-spam-to-ham))

# 3. BOGOFILTER: Scan incoming INBOX messages
# Senders in the whitelist file (one address or domain per line, # for
# comments) skip the scan, so they never land in spam.
whitelist="$(dirname "$0")/whitelist"
scan="tag:new and tag:inbox"
if [ -f "$whitelist" ]; then
    wl=""
    while IFS= read -r addr || [ -n "$addr" ]; do
        addr=$(printf '%s' "$addr" | sed 's/#.*//; s/[[:space:]"]//g')
        [ -n "$addr" ] && wl="$wl or from:\"$addr\""
    done < "$whitelist"
    [ -n "$wl" ] && scan="$scan and not (${wl# or })"
fi

# notmuch has no file: query prefix, so iterate over message ids.
notmuch search --output=messages "$scan" | while IFS= read -r id; do
    file=$(notmuch search --exclude=false --output=files -- "$id" | head -n 1)
    [ -f "$file" ] || continue
    if bogofilter < "$file"; then
        notmuch tag +spam -inbox -unread -- "$id"
    fi
done

# 4. Spam moved to Junk/Spam on another machine: once no copy is left in
# an INBOX, drop the inbox tag that this machine gave it earlier.
notmuch tag +spam -inbox -unread -- tag:inbox and \
    '(folder:cbpf/Junk or folder:"gmail/[Gmail]/Spam" or folder:proton/Spam)' and \
    not folder:cbpf/INBOX and not folder:gmail/INBOX and not folder:proton/INBOX and \
    not tag:junk-review

notmuch tag -new -- tag:new
#+end_src
